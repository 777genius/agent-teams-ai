def ident(raw):
    return {'bytes': len(raw), 'blob': hashlib.sha1(b'blob ' + str(len(raw)).encode() + b'\0' + raw).hexdigest(),
            'sha256': hashlib.sha256(raw).hexdigest()}

def digest(path):
    value = ident(path.read_bytes())
    return {'bytes': value['bytes'], 'sha256': value['sha256']}

def git(*argv):
    return subprocess.check_output(['git', *argv], cwd=repo, text=True).strip()

def git_bytes(*argv):
    return subprocess.check_output(['git', *argv], cwd=repo)

def write_json(path, value):
    with path.open('x') as stream:
        json.dump(value, stream, indent=2)
        stream.write('\n')

def sha(value):
    return isinstance(value, str) and re.fullmatch('[a-f0-9]{40}', value) is not None

proof_snapshots = {}

def frozen(spec):
    path = Path(spec['path'])
    raw = path.read_bytes()
    assert ident(raw) == {k: spec[k] for k in ('bytes', 'blob', 'sha256')}, str(path)
    return raw

def save_proof_identity(path):
    path = Path(path)
    value = ident(path.read_bytes())
    key = str(path)
    if key in proof_snapshots:
        assert proof_snapshots[key] == value, key
    proof_snapshots[key] = value
    return value

def resolved_argv(spec):
    result = [value.replace('${R1_DOCS_HEAD}', published['head'])
              .replace('${H_HEAD}', bindings['stages']['H']['head'] or '') for value in spec['argv']]
    assert not any('${' in value or value == '' for value in result)
    return result

def reference_files():
    for spec in manifest['references'].values():
        frozen(spec)
    for spec in manifest['builtOutputPolicy']['proofFiles'].values():
        frozen(spec)

def publication_state():
    doc = manifest['documentationParent']
    assert git('rev-parse', published['head'] + '^{tree}') == published['tree']
    assert git('show', '-s', '--format=%P', published['head']).split() == [doc['parentHead']]
    assert published['parents'] == [doc['parentHead']]
    changed = git('diff', '--name-only', doc['parentHead'], published['head']).splitlines()
    assert changed and all(path.startswith(doc['allowedChangedPrefix']) for path in changed)
    assert git('rev-parse', published['head'] + ':' + doc['planPath']) == published['planBlob']
    assert git('rev-parse', published['head'] + ':' + doc['unchangedPlan30Path']) == doc['unchangedPlan30Blob']
    plan = git_bytes('show', published['head'] + ':' + doc['planPath'])
    assert plan.startswith(git_bytes('cat-file', 'blob', doc['prefixBlob']))
    for key in ('canonicalR1', 'scenarioR1', 'journalR1', 'markerR1'):
        assert plan.count(frozen(manifest['references'][key])) == 1, key
    incoming = manifest['incoming']
    assert git('rev-parse', incoming['head'] + '^{tree}') == incoming['tree']
    assert git('show', '-s', '--format=%P', incoming['head']).split() == incoming['parents']

def source_state():
    reference_files()
    publication_state()
    assert git('rev-parse', 'HEAD') == expected_head
    assert git('rev-parse', 'HEAD^{tree}') == expected_tree
    expected_parents = [published['head'], manifest['incoming']['head']] if stage == 'E' else [
        bindings['stages'][chr(ord(stage) - 1)]['head']]
    assert all(sha(value) for value in expected_parents)
    assert binding['parents'] == expected_parents
    assert git('show', '-s', '--format=%P', 'HEAD').split() == expected_parents
    assert git('status', '--short') == ''
    parent = published['head'] if stage == 'E' else expected_parents[0]
    changes = git('diff', '--name-only', parent, expected_head).splitlines()
    assert sorted(changes) == sorted(manifest['stageDiffs'][stage]['paths']), changes
    tree = {}
    for line in git('ls-tree', '-r', 'HEAD').splitlines():
        fields, path = line.split('\t', 1)
        mode, kind, blob = fields.split()
        tree[path] = (mode, kind, blob)
    selected = []
    for spec in manifest['selectedSourceLeaves']:
        path = spec['path']
        actual = ident((repo / path).read_bytes())
        assert actual == spec[stage], path
        assert tree[path] == ('100644', 'blob', actual['blob']), path
        selected.append({'path': path, **actual})
    assert len(selected) == manifest['selectedSourceCount'] == 32
    integration = []
    for spec in manifest['integrationLeaves']:
        path = spec['path']
        actual = ident((repo / path).read_bytes())
        assert actual == spec[stage] and tree[path] == ('100644', 'blob', actual['blob']), path
        integration.append({'path': path, **actual})
    host_paths = sorted(path for path in tree if path.startswith('packages/legacy-agent-host/src/'))
    assert host_paths == [row['path'] for row in expected_host]
    for row in expected_host:
        path = row['path']
        assert tree[path] == (row['mode'], 'blob', row['blob'])
        assert ident((repo / path).read_bytes()) == {k: row[k] for k in ('bytes', 'blob', 'sha256')}, path
    deps = []
    for row in manifest['selectedDependencyInputs']:
        actual = {'path': row['path'], 'realpath': str((repo / row['path']).resolve()), **digest(repo / row['path'])}
        assert actual == row, row['path']
        deps.append(actual)
    assert len(deps) == manifest['dependencyCount'] == 17
    assert (repo / 'packages/legacy-agent-host/node_modules/@agent-teams/task-change-ledger').resolve() == repo / 'packages/task-change-ledger'
    for path in manifest['builtOutputPolicy']['ownedDirectories']:
        assert (repo / path).resolve() == repo / path
    return {'selectedSourceLeaves': selected, 'selectedDependencyInputs': deps, 'integrationLeaves': integration,
            'hostSourceCount': len(expected_host), 'hostSourceClosureSHA256': ident(host_raw)['sha256'],
            'clean': True, 'sourceHead': expected_head, 'sourceTree': expected_tree, 'parents': expected_parents}

def outputs():
    rows = []
    for dirname in manifest['builtOutputPolicy']['ownedDirectories']:
        directory = repo / dirname
        assert directory.resolve() == directory
        for path in sorted(directory.rglob('*')):
            assert not path.is_symlink(), str(path)
            if path.is_file():
                rows.append({'path': path.relative_to(repo).as_posix(), **digest(path)})
    return sorted(rows, key=lambda row: row['path'])

def same_d_build(current):
    policy = manifest['builtOutputPolicy']
    old_manifest = json.loads(frozen(manifest['references']['oldManifest']))
    assert old_manifest['selectedDependencyInputs'] == manifest['selectedDependencyInputs']
    assert old_manifest['builtOutputPolicy']['ownedDirectories'] == policy['ownedDirectories']
    original_source = manifest['referenceStageD']
    for family in ('ledger', 'host'):
        rec = json.loads(frozen(policy['proofFiles'][family + 'Metadata']))
        pre_raw = frozen(policy['proofFiles'][family + 'Preflight'])
        pre = json.loads(pre_raw)
        raw_log = frozen(policy['proofFiles'][family + 'Log'])
        snapshot_raw = frozen(policy['proofFiles'][family + 'Outputs'])
        saved = json.loads(snapshot_raw)
        build = next(c for c in old_manifest['commands'] if c['stage'] == 'D' and c['name'] == family + '-build')
        assert rec['sourceHead'] == original_source['head'] and rec['sourceTree'] == original_source['tree']
        assert rec['parents'] == original_source['parents'] and rec['stage'] == 'D'
        assert rec['commandName'] == 'D-' + family + '-build'
        assert rec['argv'] == build['argv'] and rec['cwd'] == build['cwd']
        assert rec['nodeSHA256'] == manifest['node']['sha256']
        assert rec['nodeVersion'] == manifest['node']['expectedVersion']
        assert rec['exitCode'] == 0 and rec['postflightExact'] is True and rec['freshSelectedOutputs'] is True
        assert rec['preflightMetadataSHA256'] == ident(pre_raw)['sha256']
        assert rec['logBytes'] == len(raw_log) and rec['logSHA256'] == ident(raw_log)['sha256']
        assert pre['sourceHead'] == original_source['head'] and pre['sourceTree'] == original_source['tree']
        assert pre['hostSourceCount'] == 615 and pre['hostSourceClosureSHA256'] == ident(host_raw)['sha256']
        assert pre['selectedDependencyInputs'] == manifest['selectedDependencyInputs']
        assert pre['selectedSourceLeaves'] == [{'path': row['path'], **row['D']} for row in old_manifest['selectedSourceLeaves']]
        assert rec['outputInventoryAfter']['bytes'] == len(snapshot_raw)
        assert rec['outputInventoryAfter']['sha256'] == ident(snapshot_raw)['sha256']
        assert len(saved) == rec['outputInventoryAfter']['count'] == policy['count'] == 1446
        assert len({row['path'] for row in saved}) == len(saved)
        if family == 'ledger':
            assert [row for row in current if row['path'].startswith('packages/task-change-ledger/')] == [
                row for row in saved if row['path'].startswith('packages/task-change-ledger/')]
        else:
            assert current == saved, 'Built outputs differ from the complete source-bound D Host inventory'
    return {'sourceHead': original_source['head'], 'sourceTree': original_source['tree'],
            'outputCount': len(current), 'reusedOnly': True,
            'outputInventory': policy['proofFiles']['hostOutputs'],
            'originalHostBuildMetadata': policy['proofFiles']['hostMetadata'],
            'originalLedgerBuildMetadata': policy['proofFiles']['ledgerMetadata']}

def read_record(record_name, expected_exit):
    previous_stage, previous_operation = record_name.split('-', 1)
    spec = next(c for c in manifest['commands'] if c['stage'] == previous_stage and c['name'] == previous_operation)
    meta_path = out / (record_name + '-metadata.json')
    rec = json.loads(meta_path.read_bytes())
    pre_path = out / (record_name + '-preflight.json')
    pre_raw = pre_path.read_bytes()
    pre = json.loads(pre_raw)
    assert rec['exitCode'] == expected_exit and rec['postflightExact'] is True
    assert rec['sourceHead'] == bindings['stages'][previous_stage]['head']
    assert rec['sourceTree'] == bindings['stages'][previous_stage]['tree']
    assert rec['commandName'] == record_name and rec['stage'] == previous_stage
    assert rec['argv'] == resolved_argv(spec) and rec['cwd'] == spec['cwd']
    assert rec['nodeSHA256'] == manifest['node']['sha256'] and rec['nodeVersion'] == manifest['node']['expectedVersion']
    assert rec['validationManifest'] == ident(manifest_raw) and rec['recorder'] == ident(recorder_raw)
    assert rec['nonBuildOutputBytesPreserved'] is True and rec['newBuildPerformed'] is False
    assert rec['preflightMetadataSHA256'] == ident(pre_raw)['sha256']
    assert pre['stageBinding'] == bindings['stages'][previous_stage] and pre['r1Published'] == published
    assert pre['validationManifest'] == ident(manifest_raw) and pre['recorder'] == ident(recorder_raw)
    assert digest(out / (record_name + '.log')) == {'bytes': rec['logBytes'], 'sha256': rec['logSHA256']}
    for path in (meta_path, pre_path, out / (record_name + '.log')):
        save_proof_identity(path)
    for filename, identity in pre['proofSnapshots'].items():
        assert ident(Path(filename).read_bytes()) == identity, filename
        save_proof_identity(filename)
    return rec

def read_review(kind, rec):
    rule = manifest['reviewRules'][kind]
    assert rule['record'] == rec['commandName']
    path = out / ('actual-' + kind + '-review.json')
    review = json.loads(path.read_bytes())
    assert review['rootApproved'] is True and review['independentApproved'] is True
    assert review['sourceHead'] == rec['sourceHead'] and review['sourceTree'] == rec['sourceTree']
    assert review['logSHA256'] == rec['logSHA256']
    assert review['metadataSHA256'] == digest(out / (rec['commandName'] + '-metadata.json'))['sha256']
    for key, expected in rule.items():
        assert review[key] == expected, kind + ': ' + key
    save_proof_identity(path)
