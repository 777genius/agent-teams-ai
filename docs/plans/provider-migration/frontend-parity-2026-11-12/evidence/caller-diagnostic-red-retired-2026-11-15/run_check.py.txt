repo = Path('/workspace/scratch/82d3f9ba6b85/agent-teams-e2e')
out = Path('/workspace/scratch/82d3f9ba6b85/caller-diagnostic-local')
review = Path('/workspace/scratch/82d3f9ba6b85/opencode-consumer-caller-r1')

commands = {

    'host-build': [node, 'node_modules/@typescript/native/bin/tsc', '-p', 'tsconfig.json'],

    'scripts-types': [node, 'node_modules/@typescript/native/bin/tsc', '--project', 'tsconfig.legacy-agent-host-scripts.json'],
    'lint': [node, 'node_modules/eslint/bin/eslint.js', '--config', 'eslint.fast.config.js', '--cache', '--cache-location', '.eslintcache-fast', '--cache-strategy', 'content', '--', *[row['path'] for row in paths]],
    'size': [node, './scripts/ci/check-source-file-size.mjs'],

}
if operation == 'ledger-build':
    cwd = repo / 'packages/task-change-ledger'
elif operation in ('host-build', 'test-types'):
    cwd = repo / 'packages/legacy-agent-host'
if operation == 'diff':
    commands[operation] = ['git', 'diff', '--check', sys.argv[3] + '..' + expected_head, '--', *[row['path'] for row in paths]]
argv = commands[operation]

def git(*args):
    return subprocess.check_output(['git', *args], cwd=repo, text=True).strip()

def digest(path):
    raw = path.read_bytes()
    return {'bytes': len(raw), 'sha256': hashlib.sha256(raw).hexdigest()}

def leaf(path):
    raw = (repo / path).read_bytes()
    blob = hashlib.sha1(b'blob ' + str(len(raw)).encode() + b'\0' + raw).hexdigest()
    assert git('rev-parse', 'HEAD:' + path) == blob
    return {'path': path, 'blob': blob, **digest(repo / path)}

def built_outputs():
    selected = ['packages/task-change-ledger/dist/index.js', 'packages/task-change-ledger/dist/index.d.ts']
    for name in ['prepareTeamModelReadiness', 'createRuntimeMemberProviderFactory']:
        selected += [f'packages/legacy-agent-host/dist/composition/{name}.{extension}' for extension in ('js', 'd.ts')]
    return [{'path': name, 'realpath': str((repo / name).resolve()), **digest(repo / name),
             'mtimeNs': (repo / name).stat().st_mtime_ns} for name in selected]

assert git('rev-parse', 'HEAD^{tree}') == expected_tree
assert git('status', '--short') == ''
version = subprocess.check_output([node, '--version'], text=True).strip()
assert version == 'v24.21.0'
node_digest = digest(Path(node))
assert node_digest['sha256'] == '7fde7b8afa198da66257f42ee2001d874c7355631e6d1579a5fb5ef1f246df4c'

dependencies = []
for original in prior['inputs']:
    current = {'path': original['path'], 'realpath': str((repo / original['path']).resolve()), **digest(repo / original['path'])}

    dependencies.append(current)
for name in ['packages/task-change-ledger/node_modules/@typescript/native/package.json',

    assert json.loads(baseline_path.read_text()) == dependencies, 'Selected dependency baseline changed'

    assert stage == 'red-ledger-build', 'The original RED build establishes the dependency baseline'
    baseline_path.write_text(json.dumps(dependencies, indent=2) + '\n')

for row in paths:
    value = leaf(row['path'])
    expected = row['candidate'] if phase == 'green' or '/tests/' in row['path'] else row['preimage']

    selected.append(value)

    selected.append(leaf(name))
