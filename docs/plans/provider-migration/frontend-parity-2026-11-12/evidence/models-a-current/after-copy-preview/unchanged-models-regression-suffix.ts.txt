
  async function heldViewRefreshFixture() {
    const requests: {
      projectPath: string | null;
      resolve: (value: RuntimeProviderManagementViewResponse) => void;
      reject: (reason: unknown) => void;
    }[] = [];
    const loadView = vi.fn((input: { projectPath?: string | null }) =>
      new Promise<RuntimeProviderManagementViewResponse>((resolve, reject) => {
        requests.push({ projectPath: input.projectPath ?? null, resolve, reject });
      })
    );
    const loadProviderDirectory = vi.fn(() => Promise.resolve(createEmptyDirectoryResponse()));
    Object.defineProperty(window, 'electronAPI', {
      configurable: true,
      value: {
        runtimeProviderManagement: { loadView, loadProviderDirectory },
      } as unknown as ElectronAPI,
    });
    const root = createRoot(host);
    await act(async () => {
      root.render(React.createElement(ConfigurableHarness, {
        enabled: true, projectPath: '/tmp/TEST-view-owner-a', loadViewOnEnable: false,
      }));
    });
    expect(loadView).not.toHaveBeenCalled();
    expect(state?.loading).toBe(false);
    const currentActions = (): RuntimeProviderManagementActions => {
      if (!actions) throw new Error('Actual hook actions were not rendered');
      return actions;
    };
    const request = (index: number) => {
      const pending = requests[index];
      if (!pending) throw new Error('Actual loadView call was not retained');
      return pending;
    };
    const complete = (index: number, model: string) => {
      const pending = request(index);
      pending.resolve({ schemaVersion: 1, runtimeId: 'opencode',
        view: { ...createRuntimeView(), projectPath: pending.projectPath, defaultModel: model } });
    };
    return { root, loadView, loadProviderDirectory, currentActions, request, complete };
  }

  it.each([
    { order: 'regular-first', outcome: 'view' },
    { order: 'silent-first', outcome: 'view' },
    { order: 'regular-first', outcome: 'response-error' },
    { order: 'silent-first', outcome: 'response-error' },
    { order: 'regular-first', outcome: 'rejection' },
    { order: 'silent-first', outcome: 'rejection' },
  ] as const)('releases current view loading after silent supersession ($order, $outcome)', async ({ order, outcome }) => {
    const fixture = await heldViewRefreshFixture();
    let regular!: Promise<boolean>, silent!: Promise<void>;
    await act(async () => { regular = fixture.currentActions().refresh(); });
    expect(state?.loading).toBe(true);
    const directoryCalls = fixture.loadProviderDirectory.mock.calls.length;
    await act(async () => { silent = fixture.currentActions().refreshDirectory(); });
    expect(fixture.loadView).toHaveBeenCalledTimes(2);
    expect(fixture.loadProviderDirectory).toHaveBeenCalledTimes(directoryCalls + 1);
    expect(fixture.request(0).projectPath).toBe('/tmp/TEST-view-owner-a');
    expect(fixture.request(1).projectPath).toBe('/tmp/TEST-view-owner-a');
    expect(state?.loading).toBe(true);
    const settled: { request: string; loading: boolean | undefined }[] = [];
    let regularResult: boolean | undefined;
    for (const request of order === 'regular-first' ? ['regular', 'silent'] : ['silent', 'regular']) {
      await act(async () => {
        if (request === 'regular') {
          fixture.complete(0, 'opencode/stale-view');
          regularResult = await regular;
        } else {
          if (outcome === 'view') fixture.complete(1, 'opencode/current-view');
          else if (outcome === 'response-error') fixture.request(1).resolve({
            schemaVersion: 1, runtimeId: 'opencode',
            error: { code: 'runtime-unhealthy', message: 'TEST current view failure', recoverable: true },
          });
          else fixture.request(1).reject(new Error('TEST current view failure'));
          await silent;
        }
      });
      settled.push({ request, loading: state?.loading });
    }
    expect(regularResult, 'the superseded regular view remains inadmissible').toBe(false);
    expect(state?.error).toBe(outcome === 'view' ? null : 'TEST current view failure');
    expect(state?.view?.defaultModel ?? null).toBe(outcome === 'view' ? 'opencode/current-view' : null);
    expect(fixture.loadView).toHaveBeenCalledTimes(2);
    if (order === 'regular-first') expect(settled[0]?.loading).toBe(true);
    expect(settled.find(row => row.request === 'silent')?.loading).toBe(false);
    expect(state?.loading).toBe(false);
  });

  it.each(['regular-first', 'silent-first'] as const)(
    'keeps a newer project view loading when both old requests settle (%s)', async order => {
      const fixture = await heldViewRefreshFixture();
      let regular!: Promise<boolean>, silent!: Promise<void>;
      await act(async () => { regular = fixture.currentActions().refresh(); });
      await act(async () => { silent = fixture.currentActions().refreshDirectory(); });
      await act(async () => {
        fixture.root.render(React.createElement(ConfigurableHarness, {
          enabled: true, projectPath: '/tmp/TEST-view-owner-b', loadViewOnEnable: false,
        }));
      });
      expect(fixture.loadView).toHaveBeenCalledTimes(3);
      expect(fixture.request(2).projectPath).toBe('/tmp/TEST-view-owner-b');
      expect(state?.loading).toBe(true);
      for (const request of order === 'regular-first' ? ['regular', 'silent'] : ['silent', 'regular']) {
        await act(async () => {
          if (request === 'regular') {
            fixture.request(0).reject(new Error('TEST stale project failure'));
            expect(await regular).toBe(false);
          } else {
            fixture.complete(1, 'opencode/stale-project');
            await silent;
          }
        });
        expect(state?.loading, 'old completions cannot release the new project request').toBe(true);
        expect(state?.view).toBeNull();
        expect(state?.error).toBeNull();
      }
      await act(async () => { fixture.complete(2, 'opencode/current-project'); });
      expect(state?.view?.projectPath).toBe('/tmp/TEST-view-owner-b');
      expect(state?.view?.defaultModel).toBe('opencode/current-project');
      expect(state?.loading).toBe(false);
      expect(fixture.loadView).toHaveBeenCalledTimes(3);
    }
  );

  it('does not publish late regular or silent view results after the actual hook unmounts', async () => {
    const fixture = await heldViewRefreshFixture();
    let regular!: Promise<boolean>, silent!: Promise<void>;
    await act(async () => { regular = fixture.currentActions().refresh(); });
    await act(async () => { silent = fixture.currentActions().refreshDirectory(); });
    const lastRenderedState = state;
    await act(async () => { fixture.root.unmount(); });
    roots.delete(fixture.root);
    await act(async () => {
      fixture.complete(1, 'opencode/after-unmount');
      await silent;
      fixture.request(0).reject(new Error('TEST stale unmounted failure'));
      expect(await regular).toBe(false);
    });
    expect(state).toBe(lastRenderedState);
    expect(fixture.loadView).toHaveBeenCalledTimes(2);
  });

  it('keeps an otherwise idle silent view refresh free of a visible loading state', async () => {
    const fixture = await heldViewRefreshFixture();
    let regular!: Promise<boolean>, silent!: Promise<void>;
    await act(async () => { regular = fixture.currentActions().refresh(); });
    await act(async () => {
      fixture.complete(0, 'opencode/initial-view');
      expect(await regular).toBe(true);
    });
    expect(state?.loading).toBe(false);
    await act(async () => { silent = fixture.currentActions().refreshDirectory(); });
    expect(state?.loading).toBe(false);
    expect(state?.view?.defaultModel).toBe('opencode/initial-view');
    await act(async () => {
      fixture.complete(1, 'opencode/refreshed-view');
      await silent;
    });
    expect(state?.loading).toBe(false);
    expect(state?.view?.defaultModel).toBe('opencode/refreshed-view');
    expect(fixture.loadView).toHaveBeenCalledTimes(2);
  });
