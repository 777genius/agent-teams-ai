import { expect, it } from 'vitest';

import { formatProviderConnectError } from '../../../../src/features/runtime-provider-management/renderer/hooks/runtimeProviderManagementPresentation';

it('explains an existing shared credential without blaming authentication or prompting a replacement', () => {
  const message = formatProviderConnectError('OpenRouter', {
    code: 'auth-failed',
    recoverable: true,
    message:
      'V2 connection cannot replace an existing OpenCode credential. Existing V1 authentication was preserved.',
  });
  expect(message).toContain('already has a saved API key');
  expect(message).toContain('Use the existing connection');
  expect(message).not.toContain('could not verify');
});

it('preserves the saved-key verification diagnostic without claiming rollback', () => {
  const detail =
    'The key was saved, but the model check failed. Check the key and connect again.';
  const message = formatProviderConnectError('test-local', {
    code: 'auth-failed',
    recoverable: true,
    message: detail,
  });
  expect(message).toBe(
    `test-local could not verify this credential with a real model request.\n${detail}`
  );
});

it('explains an invalid key without inferring whether the credential was kept', () => {
  const message = formatProviderConnectError('OpenRouter', {
    code: 'auth-failed',
    recoverable: true,
    message: 'Invalid API key',
  });
  expect(message).toBe(
    'OpenRouter rejected this API key. Copy the key from the correct account or subscription plan, then try again.'
  );
});

it('explains a policy refusal without inferring whether the credential was kept', () => {
  const message = formatProviderConnectError('OpenRouter', {
    code: 'auth-failed',
    recoverable: true,
    message: 'Access denied by security policy',
  });
  expect(message).toBe(
    'OpenRouter rejected the verification request because of an account or security policy. Check the key permissions and account restrictions, then try again.'
  );
});

it('preserves an unconfirmed operation diagnostic without inventing a storage outcome', () => {
  const detail = 'The original credential operation or model verification remains unconfirmed';
  expect(
    formatProviderConnectError('OpenRouter', {
      code: 'runtime-unhealthy',
      recoverable: false,
      message: detail,
    })
  ).toBe(detail);
});
